    @app.get("/health")
    def health() -> dict[str, str]:
        return {"status": "ok", "service": "service-observability-lab"}

    @app.get("/ready")
    def readiness(db: Session = db_dependency) -> dict[str, str]:
        db.execute(text("SELECT 1"))
        return {"status": "ready", "database": "connected"}
